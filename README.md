~~~csharp
public bool TryAssign(Hero hero) {
    if (IsCompleted) {
        return false;
    }
    if (AssignedHero != null) {
        return false;
    }
    if (!hero.IsAlive()) {
        return false;
    }
    if (hero.Health < MinHealth) {
        return false;
    }

    AssignedHero = hero;
    return true;
}
~~~
